cmake_minimum_required(VERSION 3.20)
project(deepgate_inference_sdk LANGUAGES C)

add_library(deepgate STATIC src/deepgate/dg_mfcc.c src/deepgate/dg_audio.c)
target_compile_features(deepgate PUBLIC c_std_11)
target_include_directories(deepgate PUBLIC src)
target_link_libraries(deepgate PUBLIC m)
set(DEEPGATE_MODEL_HEADER "${DEEPGATE_MODEL_HEADER}" CACHE FILEPATH "Generated deepgate_model.h for the application")
if(DEEPGATE_MODEL_HEADER)
  get_filename_component(DEEPGATE_MODEL_HEADER "${DEEPGATE_MODEL_HEADER}" ABSOLUTE)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${DEEPGATE_MODEL_HEADER}")
  file(STRINGS "${DEEPGATE_MODEL_HEADER}" model_defines
    REGEX "^#define DG_MFCC_[A-Z_]+ [0-9]+([.][0-9]+)?f?$")
  foreach(name SAMPLE_RATE N_FFT WIN HOP N_MELS N_COEFFS CLIP_HOPS TOP_DB)
    set(found FALSE)
    foreach(line IN LISTS model_defines)
      if(line MATCHES "^#define DG_MFCC_${name} (.+)$")
        target_compile_definitions(deepgate PUBLIC "DG_MFCC_${name}=${CMAKE_MATCH_1}")
        set(found TRUE)
      endif()
    endforeach()
    if(NOT found)
      message(FATAL_ERROR "Regenerate ${DEEPGATE_MODEL_HEADER}: missing DG_MFCC_${name}")
    endif()
  endforeach()
endif()

if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(deepgate PRIVATE -Wall -Wextra)
endif()

if(CMAKE_SOURCE_DIR STREQUAL PROJECT_SOURCE_DIR)
  include(CTest)
  if(BUILD_TESTING)
    enable_language(CXX)
    # Compile the actual sketch against host peripheral stubs and a shared-arena model.
    configure_file(examples/KeywordSpotting/KeywordSpotting.ino KeywordSpotting.ino COPYONLY)
    foreach(header dmadrv em_cmu em_gpio em_iadc em_prs em_timer)
      file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${header}.h" "#include \"hardware.h\"\n")
    endforeach()
    add_executable(test_sdk tests/test_sdk.cpp)
    target_compile_features(test_sdk PRIVATE cxx_std_11)
    target_include_directories(test_sdk PRIVATE ${CMAKE_CURRENT_BINARY_DIR}
      tests/arduino_stub tests/model_stub tests)
    target_link_libraries(test_sdk PRIVATE deepgate)
    foreach(group audio mfcc keywords pipeline)
      add_test(NAME ${group} COMMAND test_sdk ${group})
    endforeach()
  endif()
endif()
